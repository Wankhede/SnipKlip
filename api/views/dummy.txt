export function getEvents() {
  return async () => {
    dispatch(calendar.actions.loading());
    try {
      const response = await axios.get('/api/v3/calendar/events/');
      console.log(response.data.rows);
      // Convert JSON data response to EventInput
      // && Array.isArray(response.data.rows)
      if (response.data && response.data.rows ) {
        const eventsR: EventInput[] = response.data.rows.map((event:any) => ({
          id: event.id,
          allDay: event.allDay,
          color: event.color,
          description: event.description,
          start: new Date(event.start),
          end: new Date(event.end),
          title: event.title
        }));
      dispatch(calendar.actions.setEvents(eventsR));
      }else {
        // Handle the case when response.data.rows is not an array or is undefined
        throw new Error("Invalid response data format or missing 'rows' property");
      }
    } catch (error) {
      console.log(error);
      dispatch(calendar.actions.hasError(error));
    }
  };
}
